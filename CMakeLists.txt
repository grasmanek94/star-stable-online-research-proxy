cmake_minimum_required(VERSION 3.20)
project(SSOProxy LANGUAGES CXX)

if(NOT WIN32)
    message(FATAL_ERROR "The proxy uses Windows Winsock.")
endif()

add_executable(sso-proxy src/proxy.cpp src/hosts_redirect.cpp)
target_compile_features(sso-proxy PRIVATE cxx_std_20)
target_compile_definitions(sso-proxy PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
target_link_libraries(sso-proxy PRIVATE ws2_32)
if(MSVC)
    target_compile_options(sso-proxy PRIVATE /W4 /permissive-)
else()
    target_compile_options(sso-proxy PRIVATE -Wall -Wextra -Wpedantic)
endif()

include(CTest)
if(BUILD_TESTING)
    add_executable(proxy-hosts-test tests/hosts_redirect_test.cpp src/hosts_redirect.cpp)
    target_compile_features(proxy-hosts-test PRIVATE cxx_std_20)
    target_compile_definitions(proxy-hosts-test PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_include_directories(proxy-hosts-test PRIVATE src)
    add_test(NAME proxy-hosts-redirect COMMAND proxy-hosts-test "${CMAKE_CURRENT_BINARY_DIR}/hosts-redirect-fixture.txt")
    set_tests_properties(proxy-hosts-redirect PROPERTIES TIMEOUT 15)

    find_package(Python3 3.11 QUIET COMPONENTS Interpreter)
    if(Python3_Interpreter_FOUND)
        add_test(NAME proxy-smoke COMMAND "${Python3_EXECUTABLE}"
            "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke_proxy.py" "$<TARGET_FILE:sso-proxy>")
        set_tests_properties(proxy-smoke PROPERTIES TIMEOUT 30)
        set(tcp_smoke "${CMAKE_CURRENT_SOURCE_DIR}/tests/smoke_tcp.py")
        add_test(NAME proxy-tcp-smoke COMMAND "${Python3_EXECUTABLE}" "${tcp_smoke}" "$<TARGET_FILE:sso-proxy>")
        set_tests_properties(proxy-tcp-smoke PROPERTIES TIMEOUT 30)

        find_program(SSO_TSHARK NAMES tshark HINTS "$ENV{ProgramFiles}/Wireshark")
        find_program(SSO_OPENSSL NAMES openssl HINTS "$ENV{ProgramFiles}/Git/mingw64/bin")
        if(SSO_TSHARK AND SSO_OPENSSL)
            add_test(NAME proxy-tls-smoke COMMAND "${Python3_EXECUTABLE}" "${tcp_smoke}"
                "$<TARGET_FILE:sso-proxy>" --tshark "${SSO_TSHARK}" --openssl "${SSO_OPENSSL}")
            set_tests_properties(proxy-tls-smoke PROPERTIES TIMEOUT 45)
        endif()
    endif()
endif()
